int Bat_mode = 1;

unsigned long previousMillis = 0;
const int interval = 500;

void loop()
{
    unsigned long currentMillis = millis();

    // каждые 500 мс переключаем режим
    if (currentMillis - previousMillis >= interval)
    {
        previousMillis = currentMillis;

        Bat_mode++;

        if (Bat_mode > 7)
        {
            Bat_mode = 1;
        }
    }

    u8g2.firstPage();
    do
    {
        drawScreen_1("A", "440Hz", Bat_mode);

    } while (u8g2.nextPage());
}

void setup() {
    u8g2.begin();
}
